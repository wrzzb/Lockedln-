let timerInterval, timeLeft = 25 * 60, isRunning = false;

function updateDisplay() {
    const m = Math.floor(timeLeft / 60), s = timeLeft % 60;
    const text = `${m.toString().padStart(2,'0')}:${s.toString().padStart(2,'0')}`;
    document.getElementById('time-display').textContent = text;
    const focusDisplay = document.getElementById('focus-time-display');
    if (focusDisplay) focusDisplay.textContent = text;
}

function animateTimerButton(buttonId) {
    const btn = document.getElementById(buttonId);
    if (!btn) return;
    btn.style.transform = "scale(0.85)";
    btn.style.transition = "transform 0.1s ease";
    setTimeout(() => { btn.style.transform = ""; }, 100);
}

document.getElementById('set-custom-btn').onclick = () => {
    animateTimerButton('set-custom-btn');
    const mins = parseInt(document.getElementById('custom-minutes').value);
    if (mins > 0) { 
        clearInterval(timerInterval); 
        isRunning = false; 
        timeLeft = mins * 60; 
        updateDisplay(); 
    }
};

function startTimer() {
    if (isRunning) return;
    isRunning = true;
    timerInterval = setInterval(() => {
        if (timeLeft > 0) {
            timeLeft--;
            updateDisplay();
        } else {
            clearInterval(timerInterval);
            isRunning = false;
            document.getElementById('alarm-sound').play();
            if (focusSessionActive) endFocusSession("Time's up! Great work staying focused.");
            alert("Session completed!");
        }
    }, 1000);
}

document.getElementById('start-btn').onclick = () => {
    animateTimerButton('start-btn');
    startTimer();
};

document.getElementById('pause-btn').onclick = () => { 
    animateTimerButton('pause-btn');
    clearInterval(timerInterval); 
    isRunning = false; 
};

document.getElementById('reset-btn').onclick = () => {
    animateTimerButton('reset-btn');
    clearInterval(timerInterval); 
    isRunning = false; 
    timeLeft = 25 * 60; 
    updateDisplay();
    const alarm = document.getElementById('alarm-sound'); 
    alarm.pause(); 
    alarm.currentTime = 0;
};

// --- Focus Session ("Start Session" lock mode) ---
// Keeps the user on the Study Room page: sidebar links to other pages are
// disabled, and switching to another browser tab/window/app is tracked as
// a "distraction" (a website cannot forcibly block app/window switching).
const focusBanner = document.getElementById('focus-session-banner');
const focusBannerStatus = document.getElementById('focus-banner-status');
const startSessionBtn = document.getElementById('start-session-btn');
const endSessionBtn = document.getElementById('end-session-btn');
const distractionBadge = document.getElementById('focus-distraction-badge');
const distractionCountEl = document.getElementById('focus-distraction-count');

let focusSessionActive = false;
let distractionCount = 0;
let wakeLockHandle = null;

const FOCUS_BANNER_DEFAULT_STATUS = '<i class="fas fa-lock"></i> Focus Session Active';

async function requestWakeLock() {
    try {
        if ('wakeLock' in navigator) {
            wakeLockHandle = await navigator.wakeLock.request('screen');
        }
    } catch (err) {
        console.warn('Wake Lock unavailable:', err);
    }
}

async function releaseWakeLock() {
    if (wakeLockHandle) {
        try { await wakeLockHandle.release(); } catch (err) { /* already released */ }
        wakeLockHandle = null;
    }
}

function flagDistraction() {
    if (!focusSessionActive) return;
    distractionCount++;
    distractionCountEl.textContent = distractionCount;
    distractionBadge.classList.remove('flash');
    void distractionBadge.offsetWidth;
    distractionBadge.classList.add('flash');
}

function handleVisibilityChange() {
    if (focusSessionActive && document.hidden) flagDistraction();
}

function handleBeforeUnload(e) {
    if (!focusSessionActive) return;
    e.preventDefault();
    e.returnValue = '';
}

async function startFocusSession() {
    if (focusSessionActive) return;
    focusSessionActive = true;
    distractionCount = 0;
    distractionCountEl.textContent = '0';
    distractionBadge.classList.remove('flash');
    focusBannerStatus.innerHTML = FOCUS_BANNER_DEFAULT_STATUS;
    focusBannerStatus.classList.remove('warn');

    switchView(menuStudyBtn, studyRoomPage);
    focusBanner.classList.add('active');
    startTimer();

    await requestWakeLock();

    document.addEventListener('visibilitychange', handleVisibilityChange);
    window.addEventListener('beforeunload', handleBeforeUnload);
}

function endFocusSession(message) {
    if (!focusSessionActive) return;
    focusSessionActive = false;

    document.removeEventListener('visibilitychange', handleVisibilityChange);
    window.removeEventListener('beforeunload', handleBeforeUnload);

    focusBanner.classList.remove('active');
    releaseWakeLock();

    if (message) {
        setTimeout(() => alert(`${message}\n\nTimes you switched away: ${distractionCount}`), 250);
    }
}

if (startSessionBtn) startSessionBtn.onclick = startFocusSession;
if (endSessionBtn) endSessionBtn.onclick = () => endFocusSession("Session ended early.");

function switchView(activeMenu, activePage) {

        if (btn) btn.classList.remove('active');

}

function initDropdowns() {
    const mSelect = document.getElementById('calendar-month-select');
    const ySelect = document.getElementById('calendar-year-select');
    if (!mSelect || !ySelect) return;

    mSelect.innerHTML = ""; ySelect.innerHTML = "";

    monthNames.forEach((m, idx) => {
        const opt = document.createElement('option');
        opt.value = idx; opt.textContent = m;
        if (idx === currentMonthIndex) opt.selected = true;
        mSelect.appendChild(opt);
    });

    for (let y = 2020; y <= 2035; y++) {
        const opt = document.createElement('option');
        opt.value = y; opt.textContent = y;
        if (y === currentYear) opt.selected = true;
        ySelect.appendChild(opt);
    }

    mSelect.onchange = (e) => { currentMonthIndex = parseInt(e.target.value); renderCalendar(); };
    ySelect.onchange = (e) => { currentYear = parseInt(e.target.value); renderCalendar(); };
}

function makeDateString(d, m, y) {
    return `${d}/${String(m + 1).padStart(2, '0')}/${String(y).substring(2)}`;
}

function renderCalendar() {
    const grid = document.getElementById('calendar-days-grid');
    if (!grid) return;
    grid.innerHTML = "";

    const firstDayDate = new Date(currentYear, currentMonthIndex, 1);
    let firstDayIndex = (firstDayDate.getDay() + 6) % 7;

    const totalDays = new Date(currentYear, currentMonthIndex + 1, 0).getDate();

    for (let i = 0; i < firstDayIndex; i++) {
        const emptyCard = document.createElement('div');
        emptyCard.className = 'day-card empty-day';
        grid.appendChild(emptyCard);
    }

    const deadlinesThisMonth = userSchedules.map(s => s.deadlineString);

    for (let day = 1; day <= totalDays; day++) {
        const dayCard = document.createElement('div');
        dayCard.className = 'day-card';
        
        const loopDateStr = makeDateString(day, currentMonthIndex, currentYear);
        const todayDateStr = makeDateString(today.getDate(), today.getMonth(), today.getFullYear());

        if (loopDateStr === todayDateStr) dayCard.classList.add('today-highlight');
        if (loopDateStr === selectedDateStringGlobal) dayCard.classList.add('active-day');
        if (deadlinesThisMonth.includes(loopDateStr)) dayCard.classList.add('deadline-day');

        const dayTasks = userSchedules.filter(s => s.dateString === loopDateStr);
        const taskText = dayTasks.length > 0 ? `Task to do: ${dayTasks.length}` : 'Task to do: -';

        dayCard.innerHTML = `
            <span class="day-number">${day}</span>
            <span class="day-task-count">${taskText}</span>
        `;

        dayCard.onclick = () => {
            selectedDateStringGlobal = loopDateStr;
            document.querySelectorAll('#calendar-days-grid .day-card').forEach(card => card.classList.remove('active-day'));
            dayCard.classList.add('active-day');
            
            const selectedDateObj = new Date(currentYear, currentMonthIndex, day);
            let dayOfWeekIndex = (selectedDateObj.getDay() + 6) % 7;

            document.getElementById('selected-day-name').textContent = dayNamesShort[dayOfWeekIndex];
            document.getElementById('selected-date-string').textContent = loopDateStr;
            renderTasksList();
        };

        grid.appendChild(dayCard);
    }
}

async function loadSchedules(uid) {
    if (!uid) return;
    try {
        const res = await fetch(`${SCHEDULE_API_URL}/${uid}`);
        userSchedules = await res.json();
        renderCalendar();
        renderTasksList();
    } catch (err) {
        console.error("Failed to load schedules:", err);
    }
}

function renderTasksList() {
    const container = document.getElementById('tasks-container');
    if (!container) return;
    container.innerHTML = "";

    const activeTasks = userSchedules.filter(s => s.dateString === selectedDateStringGlobal);

    activeTasks.forEach(task => {
        const card = document.createElement('div');
        card.className = 'task-item-card';
        if (task.completed) card.classList.add('task-completed');

        card.innerHTML = `
            <div class="task-item-left">
                <div class="task-checkbox-mock">${task.completed ? '<i class="fas fa-check"></i>' : ''}</div>
                <div class="task-info-block">
                    <h4>${task.title}</h4>
                    <span class="task-tag-badge">${task.tag}</span>
                </div>
            </div>
            <div class="task-item-right">
                <span class="task-time-label">${task.time}</span>
                <span class="task-deadline-notice">Deadline: ${task.deadlineString} <i class="far fa-edit"></i></span>
            </div>
        `;

        card.querySelector('.task-checkbox-mock').onclick = async (e) => {
            e.stopPropagation();
            try {
                const res = await fetch(`${SCHEDULE_API_URL}/${task._id}`, {
                    method: 'PUT',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ completed: !task.completed })
                });
                if (res.ok) loadSchedules(auth.currentUser.uid);
            } catch (err) { console.error(err); }
        };

        card.querySelector('.task-deadline-notice').onclick = (e) => {
            e.stopPropagation();
            openEditScheduleModal(task);
        };

        container.appendChild(card);
    });
}

document.getElementById('file-upload').onchange = async (e) => {
    const user = auth.currentUser; 
    if (!user) return alert("Please log in first.");

    const label = document.querySelector('.btn-upload-header');
    const original = label.innerHTML;
    label.innerHTML = `<i class="fas fa-spinner fa-spin"></i> Uploading...`;

    const fd = new FormData(); 
    fd.append('file', file); fd.append('userId', user.uid);

    try {
        const res = await fetch('http://localhost:3000/api/upload', { method: 'POST', body: fd });
        if (res.ok) {
            loadUserMaterials(user.uid);
        } else {
            const data = await res.json().catch(() => ({}));
            const reason = data.details || data.error || res.statusText;
            alert(`Upload failed: ${reason}`);
            console.error('Upload failed:', data);
        }
    } catch (err) {
        console.error('Upload error:', err);
        alert('Upload failed: could not reach the server. Make sure the backend is running.');
    } finally { label.innerHTML = original; e.target.value = ""; }
};
